import assert from 'node:assert/strict';
import {readFile,writeFile,mkdir,copyFile} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import path from 'node:path';
import {preparePackageConsumer} from '../scripts/package-consumer.js';
import {testOptionalDependencies} from '../scripts/consumers/optional-peers.js';
import {root,run} from '../scripts/consumers/runtime.js';
import {measureMinimalBrowserBundle,BROWSER_BUNDLE_GZIP_LIMITS} from '../scripts/browser-bundle-size.js';
const fixed=JSON.parse(await readFile('.artifacts/edge-path-linked-fixed.json','utf8'));
const sha256=createHash('sha256').update(await readFile(fixed.frozenFile)).digest('hex');assert.equal(sha256,fixed.sha256);
const consumer=await preparePackageConsumer({packageSpec:fixed.frozenFile});
const receipt={node:process.version,sha256,checks:[],bundles:[]};
try{
 await testOptionalDependencies(consumer);
 await copyFile(path.join(root,'scripts/consumers/fixtures/node/edge-routes.mjs'),path.join(consumer.directory,'edge-routes.mjs'));
 await writeFile(path.join(consumer.directory,'runner.mjs'),`import {cases} from './edge-routes.mjs';import {writeFile} from 'node:fs/promises';const checks=[];for(const entry of cases.filter(value=>['edge-routes/bounded-long-paths','edge-routes/worker-authority-and-cache','edge-routes/offline-restore-and-revision','edge-routes/four-renderers-and-scene'].includes(value.id))){const start=performance.now();try{await entry.run();checks.push({id:entry.id,status:'passed',durationMs:performance.now()-start});}catch(error){checks.push({id:entry.id,status:'failed',error:error.stack});process.exitCode=1;}}await writeFile('result.json',JSON.stringify(checks));`);
 run(process.execPath,['runner.mjs'],consumer.directory,{stdio:'inherit'});receipt.checks=JSON.parse(await readFile(path.join(consumer.directory,'result.json'),'utf8'));assert.equal(receipt.checks.length,4);
 for(const specifier of ['ggaction','ggaction/basic','ggaction/svg','ggaction/graph','ggaction/animation','ggaction/interaction']){
  const measurement=await measureMinimalBrowserBundle(consumer.directory,{specifier});receipt.bundles.push(measurement);

 }
 receipt.peers=consumer.resolvedPeers;
 receipt.status=receipt.bundles.every(value=>value.gzipBytes<=BROWSER_BUNDLE_GZIP_LIMITS[value.specifier])?'passed':'bundle-limit-failed';
}finally{
 await mkdir('.artifacts/edge-path-linked-installed',{recursive:true});await writeFile('.artifacts/edge-path-linked-installed/node'+process.versions.node.split('.')[0]+'.json',JSON.stringify(receipt,null,2)+'\n');await consumer.cleanup();
}console.log(JSON.stringify(receipt,null,2));assert.equal(receipt.status,"passed");
