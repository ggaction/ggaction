import {createPackageArtifact} from '../scripts/package-artifact.js';
import path from 'node:path';
import {mkdir,copyFile,writeFile} from 'node:fs/promises';
const artifact=await createPackageArtifact();
const frozenFile=path.resolve(`.artifacts/route-job-proof-archives/ggaction-${artifact.sha256}.tgz`);
await mkdir('.artifacts/route-job-proof-archives',{recursive:true});
await copyFile(artifact.file,frozenFile);
await writeFile('.artifacts/route-job-proof-fixed.json',JSON.stringify({...artifact,frozenFile},null,2)+'\n');
console.log(JSON.stringify({...artifact,frozenFile},null,2));
