import assert from 'node:assert/strict';
import {readFile,writeFile,copyFile} from 'node:fs/promises';
import {createHash} from 'node:crypto';import path from 'node:path';
import {preparePackageConsumer} from '../scripts/package-consumer.js';
import {testOptionalDependencies} from '../scripts/consumers/optional-peers.js';
import {root,run,tscCommand} from '../scripts/consumers/runtime.js';
import {measureMinimalBrowserBundle,BROWSER_BUNDLE_GZIP_LIMITS} from '../scripts/browser-bundle-size.js';
const output='.artifacts/native-contour-calendar';const fixed=JSON.parse(await readFile(`${output}/fixed.json`,'utf8'));
const sha256=createHash('sha256').update(await readFile(fixed.frozenFile)).digest('hex');assert.equal(sha256,fixed.sha256);
const consumer=await preparePackageConsumer({packageSpec:path.resolve(fixed.frozenFile)}),receipt={node:process.version,sha256,checks:[],bundles:[]};
try{
 await testOptionalDependencies(consumer);
 await copyFile(path.join(root,'scripts/consumers/fixtures/node/contours.mjs'),path.join(consumer.directory,'contours.mjs'));
 await writeFile(path.join(consumer.directory,'runner.mjs'),`import {cases} from './contours.mjs';import {writeFile} from 'node:fs/promises';const checks=[];for(const entry of cases){const start=performance.now();try{await entry.run();checks.push({id:entry.id,status:'passed',durationMs:performance.now()-start});}catch(error){checks.push({id:entry.id,status:'failed',error:error.stack});process.exitCode=1;}}await writeFile('result.json',JSON.stringify(checks));`);
 try { run(process.execPath,['runner.mjs'],consumer.directory,{stdio:'inherit'}); } finally { receipt.checks=JSON.parse(await readFile(path.join(consumer.directory,'result.json'),'utf8')); console.log(JSON.stringify(receipt.checks)); }assert.equal(receipt.checks.length,3);
 await writeFile(path.join(consumer.directory,'contours.mts'),`import {chart,type ContourGrid} from 'ggaction';import {chart as basicChart} from 'ggaction/basic';
 const grid:ContourGrid={origin:[0,0],dx:1,dy:1,width:2,height:2,values:[0,1,1,2]};
 const p=chart().createCanvas().createContourData({id:'g',grid,thresholds:[0,1,2]}).createContourMark({mode:'line'});
 p.editContourMark({x:{reverse:true},y:{id:'next-y'},color:{id:'next-color',type:'quantize',range:['red','blue']}});
 // @ts-expect-error Basic has no contour facade
 basicChart().createContourMark();
 // @ts-expect-error intrinsic position has no arbitrary row field
 p.editContourMark({x:{field:'value'}});
 `);
 run(process.execPath,[tscCommand,'--noEmit','--strict','--exactOptionalPropertyTypes','--target','ES2022','--module','NodeNext','--moduleResolution','NodeNext','contours.mts'],consumer.directory);receipt.types='passed';
 for(const specifier of ['ggaction','ggaction/basic','ggaction/svg','ggaction/graph','ggaction/animation','ggaction/interaction'])receipt.bundles.push(await measureMinimalBrowserBundle(consumer.directory,{specifier}));
 receipt.peers=consumer.resolvedPeers;receipt.status=receipt.bundles.every(value=>value.gzipBytes<=BROWSER_BUNDLE_GZIP_LIMITS[value.specifier])?'passed':'bundle-limit-failed';
}finally{await writeFile(`${output}/node${process.versions.node.split('.')[0]}.json`,JSON.stringify(receipt,null,2)+'\n');await consumer.cleanup();}
console.log(JSON.stringify(receipt,null,2));assert.equal(receipt.status,'passed');
