import assert from 'node:assert/strict';
import {readFile,writeFile} from 'node:fs/promises';
import {join} from 'node:path';
import {preparePackageConsumer} from '../../scripts/package-consumer.js';
import {run} from '../../scripts/consumers/runtime.js';
import {testTypeScriptConsumer} from '../../scripts/consumers/types.js';
import {testOptionalDependencies} from '../../scripts/consumers/optional-peers.js';
import {measureMinimalBrowserBundle,BROWSER_BUNDLE_GZIP_LIMITS} from '../../scripts/browser-bundle-size.js';
const artifact=JSON.parse(await readFile('.artifacts/animation-v4/package.json','utf8'));
const consumer=await preparePackageConsumer({packageSpec:artifact.file});
const receipt={artifact,node:process.version,directory:consumer.directory,records:[],bundles:{}};
try {
 const file=join(consumer.directory,'public-animation.mjs');
 await writeFile(file,"import assert from 'node:assert/strict';\nimport {chart} from 'ggaction';import {chart as basic} from 'ggaction/basic';\nimport {planTransition,sampleTransition,serializeAnimation,deserializeAnimation,exportAnimationFrames,createAnimationPlayer} from 'ggaction/animation';\nconst options={durationMs:100,easing:'linear'},limits={maxEndpointNodes:200002,maxOutputNodes:100001};\nconst from=chart().createCanvas({width:200,height:160,margin:20,plotBorder:false}).createData({id:'rows',keyField:'id',values:Array.from({length:20},(_,id)=>({id,x:id/20,y:(20-id)/20}))}).createPointMark({id:'points',maxItems:20}).encodeX({field:'x',scale:{id:'x',domain:[0,1],nice:false}}).encodeY({field:'y',scale:{id:'y',domain:[0,1],nice:false}});\nconst to=from.editScale({id:'x',domain:[0,2],nice:false}),ordinary=planTransition({from,to,options});assert.equal(ordinary.version,3);\nconst native=planTransition({from,to,options:{...options,...limits}});assert.equal(native.version,4);\nconst text=serializeAnimation(native,{repeat:Infinity}),restored=deserializeAnimation(text);assert.equal(serializeAnimation(restored.plan,restored.playback),text);\nfor(const time of[0,37,100])assert.deepEqual(sampleTransition(native,time),sampleTransition(restored.plan,time));\nconst frame=sampleTransition(native,37),circles=frame.scene.nodes.filter(node=>node.type==='circle');assert.equal(circles.length,20);\nfor(let id=0;id<20;id++)assert.ok(Math.abs(circles[id].properties.x-(20+160*id/20/1.37))<1e-8);\nconst rebased=planTransition({from:frame,to,options:{...options,...limits}});assert.equal(sampleTransition(rebased,0),frame);assert.deepEqual(sampleTransition(deserializeAnimation(serializeAnimation(rebased)).plan,50),sampleTransition(rebased,50));\nassert.equal(planTransition({from:frame,to,options}).version,3);assert.throws(()=>planTransition({from,to,options:{...options,maxOutputNodes:undefined}}));\nconst b=basic().createCanvas({width:200,height:160,margin:20,plotBorder:false}).createData({id:'rows',keyField:'id',values:[{id:'a',x:.2,y:.4}]}).createPointMark({id:'points',maxItems:3}).encodeX({field:'x',scale:{id:'x',domain:[0,1],nice:false}}).encodeY({field:'y',scale:{id:'y',domain:[0,1],nice:false}});\nconst bp=planTransition({from:b,to:b,options:{...options,maxOutputNodes:100}});assert.equal(bp.version,4);assert.deepEqual(sampleTransition(deserializeAnimation(serializeAnimation(bp)).plan,37),sampleTransition(bp,37));\nconst rows=[{key:'root',parent:null,value:0},...Array.from({length:9},(_,key)=>({key,parent:'root',value:1}))];\nconst p=chart().createCanvas({width:400,height:400,margin:20,plotBorder:false}).createData({id:'rows',keyField:'key',values:rows}).createHierarchyData({id:'hierarchy',key:'key',parent:'parent',value:'value'}).layoutHierarchy({id:'layout',method:'partition'}).createPartitionPlot({id:'partition',labels:false,cells:{strokeWidth:0}}),pp=planTransition({from:p,to:p.editCanvas({width:600}),options:{...options,...limits}});\nfor(const time of[0,37,100])assert.deepEqual(sampleTransition(deserializeAnimation(serializeAnimation(pp)).plan,time),sampleTransition(pp,time));\nassert.deepEqual(exportAnimationFrames(pp,{timesMs:[0,50,100]}).map(frame=>frame.timeMs),[0,50,100]);\nawait assert.rejects(import('ggaction/animation/planner'),error=>error.code==='ERR_PACKAGE_PATH_NOT_EXPORTED');\nconsole.log('packed public version3/version4 Full/Basic/partition native/offline/rebase passed');");
 for(const [id,execute] of [
  ['optional-peers',()=>testOptionalDependencies(consumer)],
  ['public-v3-v4-native-offline-rebase',()=>run(process.execPath,[file],consumer.directory)],
  ['types',()=>testTypeScriptConsumer(consumer.directory)]
 ]){await execute();receipt.records.push({id,status:'passed'});}
 for(const specifier of Object.keys(BROWSER_BUNDLE_GZIP_LIMITS)){const measured=await measureMinimalBrowserBundle(consumer.directory,{specifier});receipt.bundles[specifier]={gzipBytes:measured.gzipBytes,limit:BROWSER_BUNDLE_GZIP_LIMITS[specifier],status:measured.gzipBytes<=BROWSER_BUNDLE_GZIP_LIMITS[specifier]?'passed':'exceeded'};}
}finally{await writeFile('.artifacts/animation-v4/consumer-'+process.versions.node+'.json',JSON.stringify(receipt,null,2)+'\n');await consumer.cleanup();}
console.log(JSON.stringify(receipt));assert.ok(Object.values(receipt.bundles).every(entry=>entry.status==='passed'));
