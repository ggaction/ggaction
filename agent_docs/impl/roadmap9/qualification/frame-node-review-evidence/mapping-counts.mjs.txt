await import('./mapping-observer.mjs');
import {writeFile} from 'node:fs/promises';
const{chart}=await import('../../src/index.js'),{planTransition,sampleTransition}=await import('../../src/animation.js');
const rows=Array.from({length:1000},(_,id)=>({id,x:1+id%9,y:(id%99+1)/100})),results=[];
for(const type of['linear','time','log','pow','sqrt','symlog']){
 const p=chart().createCanvas({width:400,height:400,margin:20,plotBorder:false}).createData({id:'rows',keyField:'id',values:rows}).createPointMark({id:'points'}).encodeX({field:'x',fieldType:type==='time'?'temporal':'quantitative',scale:{id:'x',type,domain:[1,10],nice:false}}).encodeY({field:'y',scale:{id:'y',domain:[0,1],nice:false}});
 const q=p.editScale({id:'x',domain:[1,20],nice:false}),plan=planTransition({from:p,to:q,options:{durationMs:100,easing:'linear'}});
 sampleTransition(plan,50);
 globalThis.__mappingCounts={mapping:0,inputItems:0,linear:0,pairs:0,pairItems:0,parameters:0,transformedDomain:0};const frame=sampleTransition(plan,50),counts=globalThis.__mappingCounts;delete globalThis.__mappingCounts;
 results.push({type,rows:1000,semanticPoints:frame.decisions.filter(d=>d.kernel==='cartesian-point').length,counts});
}
await writeFile(new URL('./mapping-counts.json',import.meta.url),JSON.stringify(results,null,2)+'\n');console.log(JSON.stringify(results));
