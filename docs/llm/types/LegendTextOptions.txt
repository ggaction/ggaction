  /** Gap after the occupied sample slot: categorical color/interval default 8; series 10; size/width 12.

  fontSize?: number;
  fontFamily?: string;
  fontWeight?: string | number;
  /** Continuous numeric/temporal label format. Categorical legends accept auto only. */
  format?: ValueFormat;
